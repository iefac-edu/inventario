const config = CONFIG_TABLAS[clave]; const estado = estadoTablas[clave]; const tabla = document.getElementById(config.idTabla); const datosBase = obtenerDatosVista(clave); if (datosBase.length === 0) { tabla.querySelector("thead tr").innerHTML = ""; tabla.querySelector("tbody").innerHTML = "No hay registros en esta categoría."; document.getElementById(config.idContador).textContent = ""; return; } const columnas = Object.keys(datosBase[0]); /* Encabezados con orden */ const thead = tabla.querySelector("thead tr"); thead.innerHTML = columnas.map(function (columna) { let claseOrden = ""; if (estado.ordenColumna === columna) { claseOrden = estado.ordenAsc ? "ordenAsc" : "ordenDesc"; } return "" + columna + ""; }).join(""); /* Filtrado (compara representación textual, incluyendo el resumen de diferencias) */ let filas = datosBase; if (estado.filtro) { filas = filas.filter(function (registro) { return columnas.some(function (columna) { const valorTexto = columna === "Diferencias" ? formatearValorCelda(columna, registro[columna]).replace(/<[^>]+>/g, "") : String(registro[columna] || ""); return valorTexto.toLowerCase().indexOf(estado.filtro) !== -1; }); }); } /* Orden */ if (estado.ordenColumna) { const col = estado.ordenColumna; filas = filas.slice().sort(function (a, b) { const valorA = col === "Diferencias" ? (a[col] || []).length : a[col]; const valorB = col === "Diferencias" ? (b[col] || []).length : b[col]; let comparacion; if (typeof valorA === "number" && typeof valorB === "number") { comparacion = valorA - valorB; } else { comparacion = String(valorA || "").localeCompare(String(valorB || "")); } return estado.ordenAsc ? comparacion : -comparacion; }); } /* Cuerpo */ const tbody = tabla.querySelector("tbody"); if (filas.length === 0) { tbody.innerHTML = "No se encontraron resultados."; } else { tbody.innerHTML = filas.map(function (registro) { return "" + columnas.map(function (columna) { return "" + formatearValorCelda(columna, registro[columna]) + ""; }).join("") + ""; }).join(""); } document.getElementById(config.idContador).textContent = filas.length + " de " + datosBase.length + " registros"; }
